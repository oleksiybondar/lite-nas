import { UploadDialog } from "@lite-nas/file-upload";
import SplitscreenIcon from "@mui/icons-material/Splitscreen";
import ViewAgendaIcon from "@mui/icons-material/ViewAgenda";
import UploadFileIcon from "@mui/icons-material/UploadFile";
import {
  Alert,
  Box,
  Button,
  Container,
  Dialog,
  DialogContent,
  DialogTitle,
  IconButton,
  Stack,
  Typography,
} from "@mui/material";
import { useEffect, useRef, useState } from "react";
import type { ReactElement } from "react";
 
import { ExplorerActionBar } from "./ExplorerActionBar";
import { ExplorerActionQueue } from "./ExplorerActionQueue";
import { ExplorerPanel } from "./ExplorerPanel";
import { useFileExplorer } from "./FileExplorerContext";
import { ConfirmActionDialog } from "./ConfirmActionDialog";
import { NewFolderDialog } from "./NewFolderDialog";
import { canPreviewFile, createDirectory, deleteEntry, transferEntry } from "./file-service";
import type { FileEntry } from "./file-service";
import { displayName, fileServicePath } from "./paths";
 
/** Snapshot of a consequential action while its confirmation dialog is open. */
interface PendingAction {
  kind: "copy" | "move" | "delete";
  entries: FileEntry[];
  destination?: string;
  panel: 0 | 1;
}
 
/** Two persistent browser panels and actions targeting the highlighted panel. */
export function FileExplorerWorkspace(): ReactElement {
  const { panels, activePanel, doublePanel, toggleView, queue, enqueueActions, clearSelection } =
    useFileExplorer();
  const [refresh, setRefresh] = useState(0);
  const [uploadOpen, setUploadOpen] = useState(false);
  const [previewPath, setPreviewPath] = useState<string | null>(null);
  const [error, setError] = useState("");
  const [newFolderOpen, setNewFolderOpen] = useState(false);
  const [pendingAction, setPendingAction] = useState<PendingAction | null>(null);
  const completedJobs = useRef(new Set<string>());
  const selected = panels[activePanel].selected;
  const previewable = selected.length === 1 && canPreviewFile(selected[0]);
 
  useEffect(() => {
    const finished = queue.filter(
      (item) => item.status === "done" && !completedJobs.current.has(item.id),
    );
    if (finished.length > 0) {
      for (const item of finished) completedJobs.current.add(item.id);
      setRefresh((value) => value + 1);
    }
  }, [queue]);
 
  /** Captures selected paths and the opposite panel before asking for confirmation. */
  const prepare = (kind: PendingAction["kind"]): void => {
    if (selected.length === 0) return;
    if (kind !== "delete" && !doublePanel) return;
    setPendingAction({
      kind,
      entries: [...selected],
      panel: activePanel,
      destination: kind === "delete" ? undefined : panels[activePanel === 0 ? 1 : 0].directory,
    });
  };
 
  /** Submits one queued task for each confirmed selection. */
  const confirmAction = async (): Promise<void> => {
    if (!pendingAction) return;
    const action = pendingAction;
    await enqueueActions(
      action.entries.map((entry) => ({
        label: `${action.kind} ${entry.path}`,
        run: () =>
          action.kind === "delete"
            ? deleteEntry(entry)
            : transferEntry(
                action.kind,
                entry.path,
                `${action.destination === "/" ? "" : action.destination}/${displayName(entry.path)}`,
              ),
      })),
    );
    clearSelection(action.panel);
  };
 
  /** Creates a child of the currently active directory. */
  const makeFolder = async (name: string): Promise<void> => {
    const directory = panels[activePanel].directory;
    await createDirectory(`${directory === "/" ? "" : directory}/${name}`);
    setRefresh((value) => value + 1);
  };
 
  return (
    <Container component="main" maxWidth={false} sx={{ py: 3 }} data-testid="file-explorer-page">
      <Stack spacing={2}>
        <Stack direction="row" alignItems="center" justifyContent="space-between" gap={2}>
          <Box>
            <Typography variant="h1" data-testid="file-explorer-title">
              File Explorer
            </Typography>
            <Typography variant="body2" color="text.secondary">
              {doublePanel ? "Two panels" : "One panel"} · {activePanel === 0 ? "Left" : "Right"}{" "}
              panel selected
            </Typography>
          </Box>
          <Stack direction="row" alignItems="center" gap={1}>
            <Button startIcon={<UploadFileIcon />} onClick={() => setUploadOpen(true)}>
              Upload
            </Button>
            <IconButton
              aria-label={doublePanel ? "Show one panel" : "Show two panels"}
              onClick={toggleView}
              data-testid="file-explorer-view-toggle"
            >
              {doublePanel ? <ViewAgendaIcon /> : <SplitscreenIcon />}
            </IconButton>
          </Stack>
        </Stack>
        {error && (
          <Alert severity="error" onClose={() => setError("")}>
            {error}
          </Alert>
        )}
        <Stack direction={{ xs: "column", md: "row" }} gap={2} alignItems="stretch">
          <ExplorerPanel panel={0} refresh={refresh} />
          {doublePanel && <ExplorerPanel panel={1} refresh={refresh} />}
        </Stack>
        <ExplorerActionBar
          doublePanel={doublePanel}
          hasSelection={selected.length > 0}
          canPreview={previewable}
          canDelete={selected.length > 0}
          onCopy={() => prepare("copy")}
          onMove={() => prepare("move")}
          onPreview={() => setPreviewPath(selected[0]?.path ?? null)}
          onDelete={() => prepare("delete")}
          onUpload={() => setUploadOpen(true)}
          onNewFolder={() => setNewFolderOpen(true)}
        />
        <ExplorerActionQueue />
      </Stack>
      <UploadDialog
        open={uploadOpen}
        directory={panels[activePanel].directory}
        onClose={() => setUploadOpen(false)}
        onUploaded={() => setRefresh((value) => value + 1)}
      />
      <NewFolderDialog
        open={newFolderOpen}
        directory={panels[activePanel].directory}
        onClose={() => setNewFolderOpen(false)}
        onCreate={makeFolder}
      />
      <ConfirmActionDialog
        open={pendingAction !== null}
        title={
          pendingAction
            ? `${pendingAction.kind[0].toUpperCase()}${pendingAction.kind.slice(1)} ${pendingAction.entries.length} item(s)`
            : "Confirm action"
        }
        description={
          pendingAction
            ? `Type yes to ${pendingAction.kind} ${pendingAction.entries.length} selected item(s)${pendingAction.destination ? ` to ${pendingAction.destination}` : ""}. Existing files at the destination will be replaced.`
            : ""
        }
        confirmLabel={pendingAction?.kind ?? "Confirm"}
        onClose={() => setPendingAction(null)}
        onConfirm={confirmAction}
      />
      <Dialog
        open={previewPath !== null}
        onClose={() => setPreviewPath(null)}
        fullWidth
        maxWidth="lg"
      >
        <DialogTitle>Preview</DialogTitle>
        <DialogContent>
          {previewPath !== null && (
            <iframe
              title="File preview"
              src={fileServicePath("/f", previewPath)}
              style={{ width: "100%", height: "70vh", border: 0 }}
            />
          )}
        </DialogContent>
      </Dialog>
    </Container>
  );
}
